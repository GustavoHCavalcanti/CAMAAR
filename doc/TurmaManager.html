def acesso_turma(codigo_turma)
  turma = @turmas.find { |t| t[:codigo] == codigo_turma }
  return { status: :not_found } unless turma

  if turma[:departamento] == @admin[:departamento]
    return {
      status: :success,
      turma: turma,
      pode_editar: true
    }
  else
    return {
      status: :forbidden,
      message: "A turma pertence a outro departamento",
      redirect: :lista
    }
  end
end
