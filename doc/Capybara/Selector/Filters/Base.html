def initialize(name, matcher, block, **options)
  @name = name
  @matcher = matcher
  @block = block
  @options = options
  @options[:valid_values] = [true, false] if options[:boolean]
end

def handles_option?(option_name)
  if matcher?
    @matcher.match? option_name
  else
    @name == option_name
  end
end
