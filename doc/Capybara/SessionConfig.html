def app_host=(url)
  unless url.nil? || url.match?(URI::DEFAULT_PARSER.make_regexp)
    raise ArgumentError, "Capybara.app_host should be set to a url (http://www.example.com). Attempted to set #{url.inspect}."
  end

  @app_host = url
end
