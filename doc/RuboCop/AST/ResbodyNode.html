    class RuboCop::AST::ResbodyNode

    A node extension for ‘resbody` nodes. This will be used in place of a plain node when the builder constructs the AST, making its methods available to all `resbody` nodes within RuboCop.

def exceptions
  exceptions_node = node_parts[0]
  if exceptions_node.nil?
    []
  elsif exceptions_node.array_type?
    exceptions_node.values
  else
    [exceptions_node]
  end
end
