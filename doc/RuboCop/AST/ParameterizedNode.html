    module RuboCop::AST::ParameterizedNode

    Requires implementing ‘arguments`.

    Common functionality for nodes that are parameterized: ‘send`, `super`, `zsuper`, `def`, `defs` and (modern only): `index`, `indexasgn`, `lambda`

def block_argument?

end
