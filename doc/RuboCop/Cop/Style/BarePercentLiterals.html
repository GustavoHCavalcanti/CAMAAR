    class RuboCop::Cop::Style::BarePercentLiterals

    Checks if usage of %() or %Q() matches configuration.

    @example EnforcedStyle: bare_percent (default)

    # bad
%Q(He said: "#{greeting}")
%q{She said: 'Hi'}

# good
%(He said: "#{greeting}")
%{She said: 'Hi'}

    @example EnforcedStyle: percent_q

    # bad
%|He said: "#{greeting}"|
%/She said: 'Hi'/

# good
%Q|He said: "#{greeting}"|
%q/She said: 'Hi'/
