    class RuboCop::Cop::Layout::SpaceBeforeComma

    Checks for comma (‘,`) preceded by space.

    @example

    # bad
[1 , 2 , 3]
a(1 , 2)
each { |a , b| }

# good
[1, 2, 3]
a(1, 2)
each { |a, b| }
