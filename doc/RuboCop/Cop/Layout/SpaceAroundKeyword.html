    class RuboCop::Cop::Layout::SpaceAroundKeyword

    Checks the spacing around the keywords.

    @example

    # bad
something 'test'do|x|
end

while(something)
end

something = 123if test

return(foo + bar)

# good
something 'test' do |x|
end

while (something)
end

something = 123 if test

return (foo + bar)
