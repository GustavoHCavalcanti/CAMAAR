def self.create
  supported_system = systems.find(&:supported?)
  if supported_system
    supported_system.new
  else
    puts 'RubyCritic can provide more feedback if you use ' \
         "a #{connected_system_names} repository. " \
         'Churn will not be calculated.'
    Double.new
  end
end
