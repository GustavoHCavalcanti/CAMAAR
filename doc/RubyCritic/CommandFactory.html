def self.command_class(mode)
  mode = mode.to_s.split('_').first.to_sym
  if COMMAND_CLASS_MODES.include? mode
    require "rubycritic/commands/#{mode}"
    Command.const_get(mode.capitalize)
  else
    require 'rubycritic/commands/default'
    Command::Default
  end
end
