def add_expr_debug(src, code)

end

def add_text(src, text)
  return if text.empty?
  src << (src.empty? || src[-1] == ?\n ? @indent : ' ')
  src << "_buf << "
  i = 0
  text.each_line do |line|
    src << "\n" << @indent << "        " if i > 0
    i += 1
    src << '"' << escape_text(line) << '"'
  end
  src << ";"   #<< (text[-1] == ?\n ? "\n" : "")
  src << "\n" if text[-1] == ?\n
end

def escape_text(text)
  @@table_ ||= { "\r"=>"\\r", "\n"=>"\\n", "\t"=>"\\t", '"'=>'\\"', "\\"=>"\\\\" }
  text.gsub!(/[\r\n\t"\\]/) { |m| @@table_[m] }
  return text
end
