def add_expr_debug(src, code)
  code.strip!
  s = code.gsub(/\'/, "\\'")
  src << "<?php error_log('*** debug: #{s}='.(#{code}), 0); ?>"
end

def add_stmt(src, code)
  src << "<?php"
  src << " " if code[0] != ?\ #
  if code[-1] == ?\n
    code.chomp!
    src << code << "?>\n"
  else
    src << code << "?>"
  end
end
