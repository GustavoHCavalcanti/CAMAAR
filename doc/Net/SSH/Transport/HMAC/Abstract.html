def aead(*v)
  @aead = false if !defined?(@aead)
  if v.empty?
    @aead = superclass.aead if @aead.nil? && superclass.respond_to?(:aead)
    return @aead
  elsif v.length == 1
    @aead = v.first
  else
    raise ArgumentError, "wrong number of arguments (#{v.length} for 1)"
  end
end
