def get_keys hash
  keys = Sexp.new(:array)
  hash_iterate(hash) do |key, _value|
    keys << key
  end

  keys
end

    # File vendor/bundle/ruby/3.2.0/gems/brakeman-7.1.1/lib/brakeman/processors/lib/rails2_route_processor.rb, line 291
def process_call exp
  process_default exp

  if hash? exp.target and exp.method == :keys
    keys = get_keys exp.target
    exp.clear
    keys.each_with_index do |e,i|
      exp[i] = e
    end
  end
  exp
end
