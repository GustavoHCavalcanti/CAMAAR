def inch
  @inch ||= (option ? "@option" : "@param ").tap do |text|
    text << " [Object]"
    text << (option ? " :#{source}" : " #{source}")
    text << (optional ? " (optional)" : " (required)")
    text << " #{desc}" if desc
  end
end
