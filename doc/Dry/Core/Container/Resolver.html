def call(container, key)
  item = container.fetch(key.to_s) do
    if block_given?
      return yield(key)
    else
      raise KeyError.new(%(key not found: "#{key}"), key: key.to_s, receiver: container)
    end
  end

  item.call
end

def each(container, &)
  container.map { |key, value| [key, value.call] }.each(&)
end
