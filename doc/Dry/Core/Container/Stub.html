def stub(key, value, &block)
  unless key?(key)
    raise ::ArgumentError, "cannot stub #{key.to_s.inspect} - no such key in container"
  end

  _stubs[key.to_s] = value

  if block
    yield
    unstub(key)
  end

  self
end

def unstub(*keys)
  keys = _stubs.keys if keys.empty?
  keys.each { |key| _stubs.delete(key.to_s) }
end
