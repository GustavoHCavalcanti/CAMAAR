def call(ast)
  Settings.new.tap do |settings|
    ast.each do |node|
      settings << visit(node)
    end
  end
end

def visit_nested(node)
  parent, children = node
  name, opts = parent[1]

  Setting.new(name, **opts, children: Settings.new(call(children)))
end
