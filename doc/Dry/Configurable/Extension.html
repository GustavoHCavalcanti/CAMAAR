def included(klass)
  super

  klass.class_eval do
    extend(ClassMethods)
    include(InstanceMethods)
    prepend(Initializer)

    class << self
      undef :config if method_defined?(:config)
      undef :configure if method_defined?(:configure)
    end
  end

  klass.instance_variable_set(:@__config_extension__, self)
end
