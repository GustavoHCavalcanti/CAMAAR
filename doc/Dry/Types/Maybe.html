def call_safe(input = Undefined)
  case input
  when ::Dry::Monads::Maybe
    input
  when Undefined
    None()
  else
    Maybe(type.call_safe(input) { |output = input| return yield(output) })
  end
end

def try(input = Undefined, &)
  result = type.try(input)

  if result.success?
    Result::Success.new(Maybe(result.input))
  else
    result
  end
end
