def <=>(other)
  l_path = _path
  r_path = other._path

  unless l_path.same_root?(r_path)
    raise ::ArgumentError, "Cannot compare messages from different root paths"
  end

  l_path <=> r_path
end

def to_or(root)
  clone = dup
  clone.instance_variable_set("@path", path - root.to_a)
  clone.instance_variable_set("@_path", nil)
  clone
end
