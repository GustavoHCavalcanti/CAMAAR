def call(result)
  input = result.to_h

  input_paths = key_paths(input)
  key_paths = key_map.to_dot_notation.sort

  input_paths.each do |path|
    error_path = validate_path(key_paths, path)

    next unless error_path

    result.add_error([:unexpected_key, [error_path, input]])
  end

  result
end
