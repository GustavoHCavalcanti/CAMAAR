  super()
  @config = messages.config
  @namespace = namespace
  @messages = messages
  @call_opts = {namespace: namespace}.freeze

def rule_lookup_paths(tokens)
  base_paths = messages.rule_lookup_paths(tokens)
  base_paths.map { |key|
    key.sub(config.top_namespace, "#{config.top_namespace}.#{namespace}")
  } + base_paths
end
