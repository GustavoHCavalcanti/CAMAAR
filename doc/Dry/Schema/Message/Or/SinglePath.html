def initialize(*args, messages)
  super(*args.map { [_1].flatten })
  @messages = messages

  @path = message.path

def to_or(root)
  to_ored = [left, right].map do |msgs|
    msgs.map { _1.to_or(root) }
  end

  self.class.new(*to_ored, messages)
end
