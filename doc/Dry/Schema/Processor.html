def new(options = nil)
  if options || block_given?
    processor = super(**(options || EMPTY_HASH))
    yield(processor) if block_given?
    processor
  elsif definition
    definition.call
  else
    raise ::ArgumentError, "Cannot create a schema without a definition"
  end
end
