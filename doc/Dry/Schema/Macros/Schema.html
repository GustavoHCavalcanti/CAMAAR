def call(*args, &block)
  super(*args, &nil) unless args.empty?

  if args.size.equal?(1) && (op = args.first).is_a?(::Dry::Logic::Operations::Abstract)
    process_operation(op)
  end

  if block
    schema = define(*args, &block)
    import_steps(schema)
    trace << schema.to_rule
  end

  self
end
