def extract_scope(msgid, separator)
  scope = msgid.to_s.split(separator)
  msgid = scope.pop
  [scope, msgid]
end

def plural_keys(*args)
  args.empty? ? @@plural_keys : @@plural_keys[args.first] || @@plural_keys[:en]
end
