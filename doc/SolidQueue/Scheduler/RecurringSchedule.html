def initialize(tasks)
  @configured_tasks = Array(tasks).map { |task| SolidQueue::RecurringTask.wrap(task) }.select(&:valid?)
  @scheduled_tasks = Concurrent::Hash.new
end

def schedule_tasks
  wrap_in_app_executor do
    persist_tasks
    reload_tasks
  end

  configured_tasks.each do |task|
    schedule_task(task)
  end
end
