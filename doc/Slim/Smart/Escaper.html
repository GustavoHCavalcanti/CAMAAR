def on_static(string)
  # Prevent obvious &foo; and &#1234; and &#x00ff; entities from escaping.
  block = [:multi]
  until string.empty?
    case string
    when /\A&([a-z][a-z0-9]*|#x[0-9a-f]+|#\d+);/i
      # Entity.
      block << [:escape, false, [:static, $&]]
      string = $'
    when /\A&?[^&]*/
      # Other text.
      block << [:static, $&]
      string = $'
    end
  end
  block
end
