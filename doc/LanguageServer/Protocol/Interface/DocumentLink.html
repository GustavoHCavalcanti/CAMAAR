def initialize(range:, target: nil, tooltip: nil, data: nil)
  @attributes = {}

  @attributes[:range] = range
  @attributes[:target] = target if target
  @attributes[:tooltip] = tooltip if tooltip
  @attributes[:data] = data if data

  @attributes.freeze
end
