def initialize(uri:, external: nil, take_focus: nil, selection: nil)
  @attributes = {}

  @attributes[:uri] = uri
  @attributes[:external] = external if external
  @attributes[:takeFocus] = take_focus if take_focus
  @attributes[:selection] = selection if selection

  @attributes.freeze
end
