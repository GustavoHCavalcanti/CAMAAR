def read(&block)
  while buffer = io.gets("\r\n\r\n")
    content_length = buffer.match(/Content-Length: (\d+)/i)[1].to_i
    message = io.read(content_length) or raise
    request = JSON.parse(message, symbolize_names: true)
    block.call(request)
  end
end
