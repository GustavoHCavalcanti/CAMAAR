def merge(h0, h1)

  sign = h1.delete(:sign) || 1

  h1.inject(h0) { |h, (k, v)| h.merge(k => (h[k] || 0) + sign * v) }
end

def rewrite_sdur(t)

  h = Fugit::Duration.common_rewrite_dur(t)

  sign = t.sublookup(:sign)
  sign = (sign && sign.string == '-') ? -1 : 1

  h.merge(sign: sign)
end
