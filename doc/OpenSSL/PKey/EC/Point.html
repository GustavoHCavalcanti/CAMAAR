def to_blob
  @blob ||= Net::SSH::Buffer.from(:string, ssh_type,
                                  :string, CurveNameAliasInv[group.curve_name],
                                  :mstring, to_bn.to_s(2)).to_s
  @blob
end

    Converts the key to a blob, according to the SSH2 protocol.
