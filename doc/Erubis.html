    module Erubis

    an implementation of eRuby

    ex.

    input = <<'END'
 <ul>
  <% for item in @list %>
   <li><%= item %>
       <%== item %></li>
  <% end %>
 </ul>
END
list = ['<aaa>', 'b&b', '"ccc"']
eruby = Erubis::Eruby.new(input)
puts "--- code ---"
puts eruby.src
puts "--- result ---"
context = Erubis::Context.new()   # or new(:list=>list)
context[:list] = list
puts eruby.evaluate(context)
