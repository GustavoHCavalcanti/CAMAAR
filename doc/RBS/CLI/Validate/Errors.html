def add(error)
  if error.instance_of?(WillSyntaxError)
    RBS.logger.warn(build_message(error))
    @has_syntax_error = true
  else
    @errors << error
  end
  finish if @limit == 1
end

def finish
  if @errors.empty?
    if @exit_error && @has_syntax_error
      exit 1
    else
      # success
    end
  else
    @errors.each do |error|
      RBS.logger.error(build_message(error))
    end
    exit 1
  end
end
