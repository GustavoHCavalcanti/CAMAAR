def to_s(level = 0)
  case t = type
  when RBS::Types::Literal
    case t.literal
    when Symbol
      return "#{type.to_s(1)} ?"
    end
  when RBS::Types::Proc
    return "(#{type.to_s(1)})?"
  end

  "#{type.to_s(1)}?"
end
