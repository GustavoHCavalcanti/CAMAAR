def to_s(level = 0)
  if args.empty?
    name.to_s
  else
    "#{name}[#{args.join(", ")}]"
  end
end

def with_nonreturn_void?
  each_type.any? do |type|
    if type.is_a?(Bases::Void)
      # `void` in immediate generics parameter is allowed
      false
    else
      type.with_nonreturn_void?
    end
  end
end
