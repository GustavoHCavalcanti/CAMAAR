def self.for_many_matchers(matchers)
  new(matchers.map { |m| [m.expected, diff_label_for(m), m.actual] })

def message_with_diff(message, differ)
  diff = diffs(differ)
  message = "#{message}\n#{diff}" unless diff.empty?
  message
end
