def skip_config_skip_path?(path)
  (skip_paths = CONFIG[:skip_path]) && skip_paths.any?{|skip_path| path.match?(skip_path)}
end

def skip_path?(path)
  !path ||
  DEBUGGER__.skip? ||
  ThreadClient.current.management? ||
  skip_internal_path?(path) ||
  skip_config_skip_path?(path)
end
