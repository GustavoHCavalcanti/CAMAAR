def self.config
  super do |config|
    # FIXME: Remove after Rubinius 2.0 is released
    config.datetime_format, config.datetime_proc =
      if Coercible.rbx?
        [ '%Q', Proc.new { |value| "#{value * 10**3}" } ]
      else
        [ '%s', Proc.new { |value| "#{value}" } ]
      end

    config.boolean_map = { 0 => false, 1 => true }
  end
end
