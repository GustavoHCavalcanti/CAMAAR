    Implements case-equality for the node. This is effectively == but without comparing the value of locations. Locations are checked only for presence.

    # File vendor/bundle/ruby/3.2.0/gems/prism-1.6.0/lib/prism/node.rb, line 9296
def ===(other)
  other.is_a?(IndexAndWriteNode) &&
    (flags === other.flags) &&
    (receiver === other.receiver) &&
    (call_operator_loc.nil? == other.call_operator_loc.nil?) &&
    (opening_loc.nil? == other.opening_loc.nil?) &&
    (arguments === other.arguments) &&
    (closing_loc.nil? == other.closing_loc.nil?) &&
    (block === other.block) &&
    (operator_loc.nil? == other.operator_loc.nil?) &&
    (value === other.value)
end

    # File vendor/bundle/ruby/3.2.0/gems/prism-1.6.0/lib/prism/node.rb, line 9193
def call_operator_loc
  location = @call_operator_loc
  case location
  when nil
    nil
  when Location
    location
  else
    @call_operator_loc = Location.new(source, location >> 32, location & 0xFFFFFFFF)
  end
end
