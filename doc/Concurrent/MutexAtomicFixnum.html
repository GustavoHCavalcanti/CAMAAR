def compare_and_set(expect, update)
  synchronize do
    if @value == expect.to_i
      @value = update.to_i
      true
    else
      false
    end
  end
end

def synchronize
  if @Lock.owned?
    yield
  else
    @Lock.synchronize { yield }
  end
