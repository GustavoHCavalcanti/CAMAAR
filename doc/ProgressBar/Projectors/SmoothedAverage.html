def self.calculate(current_projection, new_value, rate)
  (new_value * (1.0 - rate)) + (current_projection * rate)
end

def initialize(options = {})
  self.samples    = []
  self.projection = 0.0
  self.strength   = options[:strength] || DEFAULT_STRENGTH

  start(:at => DEFAULT_BEGINNING_POSITION)
end
