def divide_seconds(seconds)
  hours, seconds   = seconds.divmod(3600)
  minutes, seconds = seconds.divmod(60)

  [hours, minutes, seconds]
end

def elapsed_seconds
  return 0 unless started?

  ((stopped_at || time.now) - started_at)
end

def start

  self.stopped_at = nil
end
