def control_sequence_to_s(control_sequence)
  five_lsb = control_sequence.unpack('B*').first[-5..-1]
  ["000#{five_lsb}"].pack('B*')
end

def meta_char_to_codepoint(meta_char)
  byte_value = meta_char.ord
  byte_value < 128 ? byte_value + 128 : byte_value
end
