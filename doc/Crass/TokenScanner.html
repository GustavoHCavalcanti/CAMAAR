def collect
  start = @pos
  yield
  @tokens[start...@pos] || []
end

    Executes the given block, collects all tokens that are consumed during its execution, and returns them.

def consume
  @current = @tokens[@pos]
  @pos += 1 if @current
  @current
end

    Consumes the next token and returns it, advancing the pointer. Returns ‘nil` if there is no next token.
