def list_versions(*docker_args, statuses: nil)
  pipe \
    docker(:ps, *container_filter_args(statuses: statuses), *docker_args, "--format", '"{{.Names}}"'),
    extract_version_from_name

def run(hostname: nil)
  docker :run,
    "--detach",
    "--restart unless-stopped",
    "--name", container_name,
    "--network", "kamal",
    *([ "--hostname", hostname ] if hostname),
    "--env", "KAMAL_CONTAINER_NAME=\"#{container_name}\"",
    "--env", "KAMAL_VERSION=\"#{config.version}\"",
    "--env", "KAMAL_HOST=\"#{host}\"",
    "--env", "KAMAL_DESTINATION=\"#{config.destination}\"",
    *role.env_args(host),
    *role.logging_args,
    *config.volume_args,
    *role.asset_volume_args,
    *role.label_args,
    *role.option_args,
    config.absolute_image,
    role.cmd
end

def stop(version: nil)
  pipe \
    version ? container_id_for_version(version) : current_running_container_id,
    xargs(docker(:stop, *role.stop_args))
end
