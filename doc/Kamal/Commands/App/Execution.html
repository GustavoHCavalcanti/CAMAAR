def execute_in_new_container(*command, interactive: false, detach: false, env:)
  docker :run,
    (docker_interactive_args if interactive),
    ("--detach" if detach),
    ("--rm" unless detach),
    "--name", container_name_for_exec,
    "--network", "kamal",
    *role&.env_args(host),
    *argumentize("--env", env),
    *role.logging_args,
    *config.volume_args,
    *role&.option_args,
    config.absolute_image,
    *command
end
